/// Identity-bearing lifecycle result for one channel.
pub struct ChannelLifecycleResult {
    pub channel_id: String,
    pub result: Result<()>,
}

/// An owned stream and the exact admitted invocation's cancellation signal.
///
/// Adapters with their own bounded output buffer must also watch this token
/// while sending, so entry stop cannot leave them blocked on an old item.
/// It is a generation child: cancelling it never cancels a sibling invocation.
pub struct ChannelMessageStream {
    pub stream: BoxStream<'static, Result<OutboundMessage>>,
    pub cancellation: CancellationToken,
}

impl ChannelMessageStream {
    pub fn into_parts(
        self,
    ) -> (
        BoxStream<'static, Result<OutboundMessage>>,
        CancellationToken,
    ) {
        (self.stream, self.cancellation)
    }
}

impl Stream for ChannelMessageStream {
    type Item = Result<OutboundMessage>;

    fn poll_next(mut self: Pin<&mut Self>, context: &mut Context<'_>) -> Poll<Option<Self::Item>> {
        self.stream.as_mut().poll_next(context)
    }
}

fn channel_error(message: String) -> ReactError {
    ReactError::Channel(Box::new(ChannelError::Other(message)))
}

fn invocation_cancelled() -> ReactError {
    channel_error("channel handler invocation cancelled".into())
}

fn invocation_inactive(message: &str) -> ReactError {
    ReactError::Channel(Box::new(ChannelError::InvalidConfig(message.to_string())))
}

// These receipts extend the original entry, following SessionStreamReceipt's
// short synchronous admission/drop accounting. They never own another handler.
struct InvocationGeneration {
    admission: StdMutex<InvocationAdmission>,
    cancellation: CancellationToken,
    settled: Notify,
}

struct InvocationAdmission {
    ready: bool,
    active: usize,
}

impl InvocationGeneration {
    fn new() -> Self {
        Self {
            admission: StdMutex::new(InvocationAdmission {
                ready: false,
                active: 0,
            }),
            cancellation: CancellationToken::new(),
            settled: Notify::new(),
        }
    }

    fn open(&self) -> bool {
        let mut admission = self
            .admission
            .lock()
            .unwrap_or_else(|error| error.into_inner());
        if self.cancellation.is_cancelled() {
            return false;
        }
        admission.ready = true;
        true
    }

    fn retire(&self) {
        self.admission
            .lock()
            .unwrap_or_else(|error| error.into_inner())
            .ready = false;
        self.cancellation.cancel();
    }

    fn begin(self: &Arc<Self>, entry: Arc<ChannelEntry>) -> Result<InvocationReceipt> {
        let mut admission = self
            .admission
            .lock()
            .unwrap_or_else(|error| error.into_inner());
        if !admission.ready || self.cancellation.is_cancelled() {
            return Err(invocation_inactive(
                "channel handler generation is not active",
            ));
        }
        admission.active = admission
            .active
            .checked_add(1)
            .ok_or_else(|| channel_error("channel invocation capacity exhausted".into()))?;
        Ok(InvocationReceipt {
            generation: Arc::clone(self),
            _entry: entry,
        })
    }

    async fn wait(&self) {
        loop {
            let settled = self.settled.notified();
            tokio::pin!(settled);
            settled.as_mut().enable();
            if self
                .admission
                .lock()
                .unwrap_or_else(|error| error.into_inner())
                .active
                == 0
            {
                return;
            }
            settled.await;
        }
    }
}

struct InvocationReceipt {
    generation: Arc<InvocationGeneration>,
    _entry: Arc<ChannelEntry>,
}

impl Drop for InvocationReceipt {
    fn drop(&mut self) {
        let mut admission = self
            .generation
            .admission
            .lock()
            .unwrap_or_else(|error| error.into_inner());
        admission.active = admission.active.saturating_sub(1);
        if admission.active == 0 {
            self.generation.settled.notify_waiters();
        }
    }
}

struct CancelInvocationOnDrop {
    cancellation: CancellationToken,
    armed: bool,
}

impl CancelInvocationOnDrop {
    fn new(cancellation: CancellationToken) -> Self {
        Self {
            cancellation,
            armed: true,
        }
    }
    fn disarm(&mut self) {
        self.armed = false;
    }
}

impl Drop for CancelInvocationOnDrop {
    fn drop(&mut self) {
        // A cancelled waiter signals its exact child; the driver's receipt is
        // retained until local legacy drop or the driven operation truly ends.
        if self.armed {
            self.cancellation.cancel();
        }
    }
}

pub(super) async fn drive_call<T>(
    operation: impl Future<Output = Result<T>>,
    cancellation: &CancellationToken,
    caller: &CancellationToken,
    driven: bool,
) -> Result<T> {
    tokio::pin!(operation);
    if driven {
        tokio::select! {
            biased;
            () = caller.cancelled() => {
                cancellation.cancel();
                operation.await
            }
            result = &mut operation => result,
        }
    } else {
        tokio::select! {
            biased;
            () = caller.cancelled() => { cancellation.cancel(); Err(invocation_cancelled()) }
            () = cancellation.cancelled() => Err(invocation_cancelled()),
            result = &mut operation => result,
        }
    }
}

struct ChannelState {
    plugin: Box<dyn ChannelPlugin>,
    handler: Option<Arc<dyn MessageHandler>>,
    stopped_at_epoch: Option<u64>,
}

struct ChannelEntry {
    id: String,
    label: String,
    capabilities: ChannelCapabilities,
    state: RwLock<ChannelState>,
    // The original entry's current generation. No handler is stored here.
    generation: StdMutex<Option<Arc<InvocationGeneration>>>,
    mutation_epoch: AtomicU64,
    // Diagnostic projection only. All cleanup decisions use state.handler.
    unsettled_handler: AtomicBool,
}

/// Owned access to the same registered plugin and its cleanup responsibility.
///
/// Metadata is the registration-time declaration, fixed for this registration.
/// Operations retain the original instance after the manager borrow ends. A
/// callback may query manager metadata; recursively awaiting another mutation
/// of this same channel is unsupported because lifecycle operations serialize.
#[derive(Clone)]
pub struct ChannelHandle {
    entry: Arc<ChannelEntry>,
}

impl ChannelHandle {
    pub fn id(&self) -> &str {
        &self.entry.id
    }

    pub fn label(&self) -> &str {
        &self.entry.label
    }

    pub fn capabilities(&self) -> &ChannelCapabilities {
        &self.entry.capabilities
    }

    fn generation(&self) -> Option<Arc<InvocationGeneration>> {
        self.entry
            .generation
            .lock()
            .unwrap_or_else(|error| error.into_inner())
            .clone()
    }

    async fn admit(
        &self,
        captured: Option<Arc<InvocationGeneration>>,
        message_channel: &str,
    ) -> Result<(
        Arc<dyn MessageHandler>,
        InvocationReceipt,
        CancellationToken,
    )> {
        if message_channel != self.id() {
            return Err(invocation_inactive(
                "message channel_id does not match the registered entry",
            ));
        }
        let generation =
            captured.ok_or_else(|| invocation_inactive("channel handler has not started"))?;
        if !self
            .generation()
            .is_some_and(|current| Arc::ptr_eq(&current, &generation))
        {
            return Err(ReactError::Channel(Box::new(
                ChannelError::StaleInvocation {
                    channel_id: self.id().to_string(),
                },
            )));
        }
        // Fail before awaiting the lifecycle lock during start/rollback/stop.
        let ready = generation
            .admission
            .lock()
            .unwrap_or_else(|error| error.into_inner())
            .ready;
        if !ready || generation.cancellation.is_cancelled() {
            return Err(invocation_inactive(
                "channel handler generation is not active",
            ));
        }
        let state = self.entry.state.read().await;
        if !self
            .generation()
            .is_some_and(|current| Arc::ptr_eq(&current, &generation))
        {
            return Err(ReactError::Channel(Box::new(
                ChannelError::StaleInvocation {
                    channel_id: self.id().to_string(),
                },
            )));
        }
        let handler = state
            .handler
            .as_ref()
            .cloned()
            .ok_or_else(|| invocation_inactive("channel handler is unavailable"))?;
        let receipt = generation.begin(Arc::clone(&self.entry))?;
        let cancel = generation.cancellation.child_token();
        Ok((handler, receipt, cancel))
    }

    /// Invoke the original successfully started handler with a caller-local token.
    /// Dropping the waiter signals cancellation; shutdown still waits its driver.
    pub fn handle_message(
        &self,
        message: InboundMessage,
        caller: CancellationToken,
    ) -> BoxFuture<'static, Result<OutboundMessage>> {
        let channel = self.clone();
        let generation = self.generation();
        Box::pin(async move {
            let runtime = tokio::runtime::Handle::try_current().map_err(|error| {
                channel_error(format!("channel invocation requires a runtime: {error}"))
            })?;
            if caller.is_cancelled() {
                return Err(invocation_cancelled());
            }
            let (handler, receipt, cancellation) =
                channel.admit(generation, &message.channel_id).await?;
            let mut cancel_on_drop = CancelInvocationOnDrop::new(cancellation.clone());
            let (sender, receiver) = oneshot::channel();
            let driver = runtime.spawn(async move {
                let _receipt = receipt;
                let result = drive_call(
                    handler.handle_with_cancel(message, cancellation.clone()),
                    &cancellation,
                    &caller,
                    handler.settles_on_cancel(),
                )
                .await;
                let _ = sender.send(result);
            });
            let result = receiver.await;
            driver.await.map_err(|error| {
                channel_error(format!("channel handler driver failed: {error}"))
            })?;
            cancel_on_drop.disarm();
            result
                .map_err(|error| channel_error(format!("channel handler driver ended: {error}")))?
        })
    }

    /// Invoke reply on the same original handler; it is distinct from plugin.send.
    pub fn reply_message(
        &self,
        message: OutboundMessage,
        caller: CancellationToken,
    ) -> BoxFuture<'static, Result<()>> {
        let channel = self.clone();
        let generation = self.generation();
        Box::pin(async move {
            let runtime = tokio::runtime::Handle::try_current().map_err(|error| {
                channel_error(format!("channel invocation requires a runtime: {error}"))
            })?;
            if caller.is_cancelled() {
                return Err(invocation_cancelled());
            }
            let (handler, receipt, cancellation) =
                channel.admit(generation, &message.channel_id).await?;
            let mut cancel_on_drop = CancelInvocationOnDrop::new(cancellation.clone());
            let (sender, receiver) = oneshot::channel();
            let driver = runtime.spawn(async move {
                let _receipt = receipt;
                let result = drive_call(
                    handler.reply_with_cancel(message, cancellation.clone()),
                    &cancellation,
                    &caller,
                    handler.settles_on_cancel(),
                )
                .await;
                let _ = sender.send(result);
            });
            let result = receiver.await;
            driver
                .await
                .map_err(|error| channel_error(format!("channel reply driver failed: {error}")))?;
            cancel_on_drop.disarm();
            result.map_err(|error| channel_error(format!("channel reply driver ended: {error}")))?
        })
    }

    /// Return an owned stream tied to the captured original handler generation.
    /// Its bounded driver owns the trait's borrowed inner stream and receipt.
    pub fn handle_message_stream(
        &self,
        message: InboundMessage,
        caller: CancellationToken,
    ) -> BoxFuture<'static, Result<ChannelMessageStream>> {
        let channel = self.clone();
        let generation = self.generation();
        Box::pin(async move {
            let runtime = tokio::runtime::Handle::try_current().map_err(|error| {
                channel_error(format!("channel invocation requires a runtime: {error}"))
            })?;
            if caller.is_cancelled() {
                return Err(invocation_cancelled());
            }
            let (handler, receipt, cancellation) =
                channel.admit(generation, &message.channel_id).await?;
            let invocation_cancellation = cancellation.clone();
            let cancel_on_drop = CancelInvocationOnDrop::new(cancellation.clone());
            let (setup_sender, setup_receiver) = oneshot::channel();
            let (sender, mut receiver) = mpsc::channel(1);
            let driver = runtime.spawn(async move {
                let _receipt = receipt;
                let driven = handler.settles_on_cancel();
                let setup = drive_call(
                    handler.handle_stream_with_cancel(message, cancellation.clone()),
                    &cancellation,
                    &caller,
                    driven,
                )
                .await;
                let mut inner = match setup {
                    Ok(inner) => inner,
                    Err(error) => {
                        let _ = setup_sender.send(Err(error));
                        return;
                    }
                };
                if setup_sender.send(Ok(())).is_err() {
                    cancellation.cancel();
                }
                loop {
                    let item = drive_call(
                        async { Ok(inner.next().await) },
                        &cancellation,
                        &caller,
                        driven,
                    )
                    .await;
                    let item = match item {
                        Ok(Some(item)) => item,
                        Ok(None) | Err(_) => break,
                    };
                    // Driven cancellation continues polling to actual completion;
                    // it never blocks on an abandoned/full consumer buffer.
                    if cancellation.is_cancelled() {
                        continue;
                    }
                    tokio::select! {
                        biased;
                        () = caller.cancelled() => { cancellation.cancel(); }
                        () = cancellation.cancelled() => {},
                        sent = sender.send(item) => { if sent.is_err() { cancellation.cancel(); } }
                    }
                }
                drop(inner);
            });
            setup_receiver.await.map_err(|error| {
                channel_error(format!("channel stream driver ended during setup: {error}"))
            })??;
            let stream = async_stream::stream! {
                let mut cancel_on_drop = cancel_on_drop;
                loop {
                    let item = tokio::select! {
                        biased;
                        () = cancel_on_drop.cancellation.cancelled() => break,
                        item = receiver.recv() => item,
                    };
                    let Some(item) = item else { break };
                    yield item;
                }
                let exit = driver.await;
                cancel_on_drop.disarm();
                if let Err(error) = exit {
                    yield Err(channel_error(format!("channel stream driver failed: {error}")));
                }
            };
            Ok(ChannelMessageStream {
                stream: Box::pin(stream),
                cancellation: invocation_cancellation,
            })
        })
    }

    fn invalidate_stopped(&self) {
        // MAX remains permanently conservative, including concurrent invalidation
        // while stop awaits a plugin callback. Never wrap into an old epoch.
        let mut epoch = self.entry.mutation_epoch.load(Ordering::Acquire);
        loop {
            match self.entry.mutation_epoch.compare_exchange_weak(
                epoch,
                epoch.saturating_add(1),
                Ordering::AcqRel,
                Ordering::Acquire,
            ) {
                Ok(_) => break,
                Err(current) => epoch = current,
            }
        }
    }

    async fn start_with(
        &self,
        handler_factory: impl FnOnce() -> Arc<dyn MessageHandler> + Send,
    ) -> Result<()> {
        let mut state = self.entry.state.write().await;
        if state.handler.is_some() {
            return Err(channel_error(format!(
                "Channel '{}' still owns a handler; stop it before restarting",
                self.id()
            )));
        }
        state.stopped_at_epoch = None;
        let generation = Arc::new(InvocationGeneration::new());
        *self
            .entry
            .generation
            .lock()
            .unwrap_or_else(|error| error.into_inner()) = Some(Arc::clone(&generation));
        let handler = handler_factory();
        state.handler = Some(Arc::clone(&handler));
        self.entry.unsettled_handler.store(true, Ordering::Release);
        match state.plugin.start(Arc::clone(&handler)).await {
            Ok(()) => {
                // Transport start succeeded even if a concurrent stop already
                // retired admission. The queued stop retains cleanup ownership;
                // preserve the original start result without reopening the gate.
                generation.open();
                Ok(())
            }
            Err(start_error) => {
                generation.retire();
                // Publish responsibility before the first plugin await. Future
                // cancellation leaves that same handler available to stop.
                let epoch = self.entry.mutation_epoch.load(Ordering::Acquire);
                match state.plugin.stop().await {
                    Ok(()) => {
                        state.stopped_at_epoch = Some(epoch);
                        match handler.close().await {
                            Ok(()) => {
                                state.handler = None;
                                self.entry.unsettled_handler.store(false, Ordering::Release);
                                Err(start_error)
                            }
                            Err(close_error) => Err(channel_error(format!(
                                "Channel '{}' start failed: {start_error}; handler close failed: {close_error}",
                                self.id()
                            ))),
                        }
                    }
                    Err(stop_error) => Err(channel_error(format!(
                        "Channel '{}' start failed: {start_error}; rollback stop failed: {stop_error}",
                        self.id()
                    ))),
                }
            }
        }
    }

    /// Start using the original plugin, retaining handler responsibility on failure or cancellation.
    pub fn start(&self, handler: Arc<dyn MessageHandler>) -> BoxFuture<'static, Result<()>> {
        let channel = self.clone();
        Box::pin(async move { channel.start_with(|| handler).await })
    }

    /// Await transport stop and handler close. Retry does not repeat a confirmed stop.
    pub fn stop(&self) -> BoxFuture<'static, Result<()>> {
        let channel = self.clone();
        Box::pin(async move {
            if let Some(generation) = channel.generation() {
                generation.retire();
            }
            let mut state = channel.entry.state.write().await;
            let generation = channel.generation();
            if let Some(generation) = &generation {
                generation.retire();
            }
            let epoch = channel.entry.mutation_epoch.load(Ordering::Acquire);
            if epoch == u64::MAX || state.stopped_at_epoch != Some(epoch) {
                state.plugin.stop().await?;
                state.stopped_at_epoch = Some(epoch);
            }
            if let Some(handler) = &state.handler {
                if let Some(generation) = &generation {
                    generation.wait().await;
                }
                handler.close().await?;
                state.handler = None;
                channel
                    .entry
                    .unsettled_handler
                    .store(false, Ordering::Release);
            }
            Ok(())
        })
    }

    pub fn send(&self, message: OutboundMessage) -> BoxFuture<'static, Result<()>> {
        let channel = self.clone();
        Box::pin(async move { channel.entry.state.read().await.plugin.send(message).await })
    }

    pub fn health_check(&self) -> BoxFuture<'static, Result<()>> {
        let channel = self.clone();
        Box::pin(async move { channel.entry.state.read().await.plugin.health_check().await })
    }
}

#[async_trait]
impl ChannelPlugin for ChannelHandle {
    fn id(&self) -> &str {
        self.id()
    }
    fn label(&self) -> &str {
        self.label()
    }
    fn capabilities(&self) -> &ChannelCapabilities {
        self.capabilities()
    }
    async fn start(&mut self, handler: Arc<dyn MessageHandler>) -> Result<()> {
        ChannelHandle::start(self, handler).await
    }
    async fn stop(&mut self) -> Result<()> {
        ChannelHandle::stop(self).await
    }
    async fn send(&self, message: OutboundMessage) -> Result<()> {
        ChannelHandle::send(self, message).await
    }
    async fn health_check(&self) -> Result<()> {
        ChannelHandle::health_check(self).await
    }
}

/// Registry and lifecycle access for IM channel plugins.
///
/// `id`, `label` and `capabilities` are fixed registration declarations. The old
/// borrowed accessors delegate to the same entry as owned access; they do not
/// expose later dynamic changes to a third-party plugin's metadata.
/// Obtain an owned operation under an external manager lock, release that lock,
/// then await. Explicit awaited shutdown is required; dropping is not shutdown.
pub struct ChannelManager {
    channels: HashMap<String, ChannelHandle>,
}

impl Default for ChannelManager {
    fn default() -> Self {
        Self::new()
    }
}

impl ChannelManager {
    pub fn new() -> Self {
        Self {
            channels: HashMap::new(),
        }
    }

    /// Register the original plugin with its fixed metadata declaration.
    pub fn register(&mut self, plugin: Box<dyn ChannelPlugin>) -> Result<()> {
        let id = plugin.id().to_string();
        if self.channels.contains_key(&id) {
            return Err(channel_error(format!(
                "Channel '{id}' is already registered"
            )));
        }
        let capabilities = plugin.capabilities();
        let capabilities = ChannelCapabilities {
            chat_types: capabilities.chat_types,
            supports_media: capabilities.supports_media,
            supports_threads: capabilities.supports_threads,
        };
        let entry = Arc::new(ChannelEntry {
            id: id.clone(),
            label: plugin.label().to_string(),
            capabilities,
            state: RwLock::new(ChannelState {
                plugin,
                handler: None,
                stopped_at_epoch: None,
            }),
            generation: StdMutex::new(None),
            mutation_epoch: AtomicU64::new(0),
            unsettled_handler: AtomicBool::new(false),
        });
        info!("Registering channel: {}", id);
        self.channels.insert(id, ChannelHandle { entry });
        Ok(())
    }

    pub fn len(&self) -> usize {
        self.channels.len()
    }
    pub fn is_empty(&self) -> bool {
        self.channels.is_empty()
    }

    /// Retain the original entry independently of the manager borrow.
    pub fn get_owned(&self, id: &str) -> Option<ChannelHandle> {
        self.channels.get(id).cloned()
    }

    fn snapshot(&self) -> Vec<ChannelHandle> {
        self.channels.values().cloned().collect()
    }

    async fn start_entries(
        entries: Vec<ChannelHandle>,
        handler_factory: impl Fn(&str) -> Arc<dyn MessageHandler> + Sync,
    ) -> Vec<ChannelLifecycleResult> {
        let mut results = Vec::with_capacity(entries.len());
        for channel in entries {
            let result = channel.start_with(|| handler_factory(channel.id())).await;
            results.push(ChannelLifecycleResult {
                channel_id: channel.id().to_string(),
                result,
            });
        }
        results
    }

    /// Capture current registrations now; invoke the factory only when polled.
    /// Channels registered later are considered by the next batch.
    pub fn start_all_owned(
        &self,
        handler_factory: impl Fn(&str) -> Arc<dyn MessageHandler> + Send + Sync + 'static,
    ) -> BoxFuture<'static, Vec<ChannelLifecycleResult>> {
        let entries = self.snapshot();
        Box::pin(Self::start_entries(entries, handler_factory))
    }

    /// Start the current channels sequentially and retain per-channel results.
    pub async fn start_all(
        &mut self,
        handler_factory: impl Fn(&str) -> Arc<dyn MessageHandler> + Sync,
    ) -> Vec<ChannelLifecycleResult> {
        Self::start_entries(self.snapshot(), &handler_factory).await
    }

    pub fn stop_owned(&self, channel_id: &str) -> BoxFuture<'static, Result<()>> {
        let channel = self.get_owned(channel_id);
        let id = channel_id.to_string();
        Box::pin(async move {
            channel
                .ok_or_else(|| channel_error(format!("Channel '{id}' not found")))?
                .stop()
                .await
        })
    }

    pub async fn stop(&mut self, channel_id: &str) -> Result<()> {
        self.stop_owned(channel_id).await
    }

    /// Capture current registrations now and attempt every channel on poll.
    pub fn stop_all_owned(&self) -> BoxFuture<'static, Result<()>> {
        let entries = self.snapshot();
        Box::pin(async move {
            let mut failures = Vec::new();
            for channel in entries {
                if let Err(error) = channel.stop().await {
                    warn!("Failed to stop channel '{}': {}", channel.id(), error);
                    failures.push(format!("{}: {error}", channel.id()));
                }
            }
            if failures.is_empty() {
                Ok(())
            } else {
                Err(channel_error(format!(
                    "failed to stop channels: {}",
                    failures.join("; ")
                )))
            }
        })
    }

    pub async fn stop_all(&mut self) -> Result<()> {
        self.stop_all_owned().await
    }

    pub fn get(&self, id: &str) -> Option<&(dyn ChannelPlugin + '_)> {
        self.channels
            .get(id)
            .map(|channel| channel as &dyn ChannelPlugin)
    }

    /// Mutable borrowed access conservatively invalidates a prior stop fact.
    /// Availability does not change when an owned handle is also retained.
    pub fn get_mut(&mut self, id: &str) -> Option<&mut (dyn ChannelPlugin + '_)> {
        match self.channels.get_mut(id) {
            Some(channel) => {
                channel.invalidate_stopped();
                Some(channel as &mut dyn ChannelPlugin)
            }
            None => None,
        }
    }

    pub fn channel_ids(&self) -> Vec<&str> {
        self.channels.keys().map(String::as_str).collect()
    }
}

impl Drop for ChannelManager {
    fn drop(&mut self) {
        let unsettled = self
            .channels
            .values()
            .filter(|channel| channel.entry.unsettled_handler.load(Ordering::Acquire))
            .count();
        if unsettled != 0 {
            info!(
                "ChannelManager dropped with {} unsettled handlers remaining; await stop_all() for shutdown",
                unsettled
            );
        }
    }
}

impl Drop for ChannelEntry {
    fn drop(&mut self) {
        if self.state.get_mut().handler.is_some() {
            warn!(
                "Channel '{}' final owner dropped with an unsettled handler; shutdown was not awaited",
                self.id
            );
        }
    }
}

    #[derive(Default)]
    struct Facts {
        calls: AtomicUsize,
        replies: AtomicUsize,
        closes: AtomicUsize,
        drops: AtomicUsize,
        entered: Notify,
        cancelled: Notify,
    }

    struct DropFact(Arc<Facts>);
    impl Drop for DropFact {
        fn drop(&mut self) {
            self.0.drops.fetch_add(1, Ordering::AcqRel);
        }
    }

    struct ForwardHandler {
        facts: Arc<Facts>,
        driven: bool,
        setup_error: bool,
        close_error_once: bool,
        release: Arc<Semaphore>,
    }

    impl ForwardHandler {
        fn plain(facts: Arc<Facts>) -> Self {
            Self {
                facts,
                driven: false,
                setup_error: false,
                close_error_once: false,
                release: Arc::new(Semaphore::new(0)),
            }
        }

        async fn run(
            &self,
            message: InboundMessage,
            cancel: CancellationToken,
        ) -> Result<OutboundMessage> {
            self.facts.calls.fetch_add(1, Ordering::AcqRel);
            let _drop = DropFact(Arc::clone(&self.facts));
            if message.text == "park" {
                self.facts.entered.notify_one();
                cancel.cancelled().await;
                self.facts.cancelled.notify_one();
                if self.driven {
                    self.release
                        .acquire()
                        .await
                        .map_err(|error| channel_error(error.to_string()))?
                        .forget();
                }
                return Err(invocation_cancelled());
            }
            Ok(OutboundMessage::new(
                message.channel_id,
                message.chat_id,
                message.chat_type,
                message.text,
            ))
        }
    }

    #[async_trait]
    impl MessageHandler for ForwardHandler {
        async fn handle(&self, message: InboundMessage) -> Result<OutboundMessage> {
            self.run(message, CancellationToken::new()).await
        }
        async fn handle_with_cancel(
            &self,
            message: InboundMessage,
            cancel: CancellationToken,
        ) -> Result<OutboundMessage> {
            self.run(message, cancel).await
        }
        async fn reply(&self, _message: OutboundMessage) -> Result<()> {
            self.facts.replies.fetch_add(1, Ordering::AcqRel);
            Ok(())
        }
        fn settles_on_cancel(&self) -> bool {
            self.driven
        }
        async fn close(&self) -> Result<()> {
            let n = self.facts.closes.fetch_add(1, Ordering::AcqRel);
            if self.close_error_once && n == 0 {
                return Err(channel_error("retry close".into()));
            }
            Ok(())
        }
        async fn handle_stream_with_cancel<'a>(
            &'a self,
            message: InboundMessage,
            cancel: CancellationToken,
        ) -> Result<BoxStream<'a, Result<OutboundMessage>>> {
            if self.setup_error {
                return Err(channel_error("setup failed".into()));
            }
            if message.text == "setup-park" {
                let _drop = DropFact(Arc::clone(&self.facts));
                self.facts.entered.notify_one();
                cancel.cancelled().await;
                return Err(invocation_cancelled());
            }
            let stream = async_stream::stream! {
                let _drop = DropFact(Arc::clone(&self.facts));
                self.facts.entered.notify_one();
                if message.text == "park" {
                    cancel.cancelled().await;
                    self.facts.cancelled.notify_one();
                    if self.driven
                        && let Err(error) = self.release.acquire().await.map(|permit| permit.forget()) {
                        yield Err(channel_error(error.to_string()));
                    }
                } else {
                    for text in ["第一段😀", "second", "third"] {
                        yield Ok(OutboundMessage::new(&message.channel_id, &message.chat_id, message.chat_type, text));
                    }
                }
            };
            Ok(Box::pin(stream))
        }
    }

    struct ForwardPlugin {
        fail_start: bool,
        fail_stop: bool,
        start_gate: Option<(Arc<Notify>, Arc<Semaphore>)>,
    }

    #[async_trait]
    impl ChannelPlugin for ForwardPlugin {
        fn id(&self) -> &str {
            "forward"
        }
        fn capabilities(&self) -> &ChannelCapabilities {
            static CAPS: ChannelCapabilities = ChannelCapabilities {
                chat_types: &[ChatType::Direct],
                supports_media: true,
                supports_threads: false,
            };
            &CAPS
        }
        async fn start(&mut self, _handler: Arc<dyn MessageHandler>) -> Result<()> {
            if let Some((entered, gate)) = &self.start_gate {
                entered.notify_one();
                gate.acquire()
                    .await
                    .map_err(|error| channel_error(error.to_string()))?
                    .forget();
            }
            if self.fail_start {
                return Err(channel_error("start failed".into()));
            }
            Ok(())
        }
        async fn stop(&mut self) -> Result<()> {
            if self.fail_stop {
                self.fail_stop = false;
                return Err(channel_error("stop failed".into()));
            }
            Ok(())
        }
        async fn send(&self, _message: OutboundMessage) -> Result<()> {
            Ok(())
        }
    }

        let channel = started(session.clone()).await?;
        let first_cancel = CancellationToken::new();
        let first = tokio::spawn(channel.handle_message(message("park"), first_cancel.clone()));
        timeout(LIMIT, facts.entered.notified()).await?;
        let old = instances
            .lock()
            .unwrap_or_else(|error| error.into_inner())
            .first()
            .cloned()
            .ok_or("missing instance")?;
        let reset_cancel = CancellationToken::new();
        let reset_token = reset_cancel.clone();
        let reset_session = session.clone();
        let mut reset = Box::pin(async move {
            if stream_reset {
                reset_session
                    .handle_stream_with_cancel(message("/reset"), reset_token)
                    .await
                    .map(|_| ())
            } else {
                reset_session
                    .handle_with_cancel(message("/reset"), reset_token)
                    .await
                    .map(|_| ())
            }
        });
        // Poll B into A's held sender lock deterministically before cancelling;
        // no scheduler sleep is used as evidence that the reset was queued.
        assert!(futures::poll!(reset.as_mut()).is_pending());
        reset_cancel.cancel();
        first_cancel.cancel();
        timeout(LIMIT, facts.cancelled.notified()).await?;
        release.add_permits(1);
        assert!(timeout(LIMIT, first).await??.is_err());
        assert!(timeout(LIMIT, reset).await?.is_err());
        assert_eq!(
            instances
                .lock()
                .unwrap_or_else(|error| error.into_inner())
                .as_slice(),
            &[old]
        );
        assert_eq!(facts.closes.load(Ordering::Acquire), 0);
        assert_eq!(session.active_sessions(), 1);
        assert_eq!(
            channel
                .handle_message(message("same incarnation"), CancellationToken::new())
                .await?
                .text,
            "same incarnation"
        );
        channel.stop().await?;
        Ok(())
    }

    struct ClosingHandler {
        stopped: Arc<AtomicBool>,
        closes: Arc<AtomicUsize>,
    }

    #[async_trait]
    impl MessageHandler for ClosingHandler {
        async fn handle(&self, _msg: InboundMessage) -> Result<OutboundMessage> {
            Err(ReactError::Other("not used".to_string()))
        }

        async fn reply(&self, _msg: OutboundMessage) -> Result<()> {
            Ok(())
        }

        async fn close(&self) -> Result<()> {
            if !self.stopped.load(Ordering::Acquire) {
                return Err(ReactError::Other("transport still accepting".to_string()));
            }
            let attempt = self.closes.fetch_add(1, Ordering::AcqRel);
            if attempt == 0 {
                Err(ReactError::Other("transient close failure".to_string()))
            } else {
                Ok(())
            }
        }
    }

    struct ClosingPlugin {
        stopped: Arc<AtomicBool>,
        stops: Arc<AtomicUsize>,
        fail_repeated_stop: bool,
        handler: Option<Arc<dyn MessageHandler>>,
    }

    #[async_trait]
    impl ChannelPlugin for ClosingPlugin {
        fn id(&self) -> &str {
            "close-test"
        }

        fn capabilities(&self) -> &ChannelCapabilities {
            static CAPABILITIES: ChannelCapabilities = ChannelCapabilities {
                chat_types: &[ChatType::Direct],
                supports_media: false,
                supports_threads: false,
            };
            &CAPABILITIES
        }

        async fn start(&mut self, handler: Arc<dyn MessageHandler>) -> Result<()> {
            self.stopped.store(false, Ordering::Release);
            self.handler = Some(handler);
            Ok(())
        }

        async fn stop(&mut self) -> Result<()> {
            let attempt = self.stops.fetch_add(1, Ordering::AcqRel);
            if self.fail_repeated_stop && attempt > 0 {
                return Err(ReactError::Other("transport already stopped".to_string()));
            }
            self.stopped.store(true, Ordering::Release);
            self.handler = None;
            Ok(())
        }

        async fn send(&self, _msg: OutboundMessage) -> Result<()> {
            Err(ReactError::Channel(Box::new(ChannelError::Other(
                "not used".to_string(),
            ))))
        }
    }

    #[tokio::test]

        let stops = Arc::new(AtomicUsize::new(0));
        let closes = Arc::new(AtomicUsize::new(0));

                    closes: Arc::clone(&closes),
                })

        assert!(results.iter().all(|result| result.result.is_ok()));

        let first = manager.stop("close-test").await;
        assert!(first.is_err());
        assert_eq!(closes.load(Ordering::Acquire), 1);
        manager.stop("close-test").await?;

#[cfg(test)]
mod owned_tests {
    use super::*;
    use crate::channels::types::{ChatType, InboundMessage};
    use std::sync::Weak;
    use std::sync::atomic::AtomicUsize;
    use std::time::Duration;
    use tokio::sync::{Mutex, Notify};

    const CAPS: ChannelCapabilities = ChannelCapabilities {
        chat_types: &[ChatType::Direct],
        supports_media: false,
        supports_threads: false,
    };

    struct Probe {
        manager: Weak<Mutex<ChannelManager>>,
        calls: Mutex<Vec<&'static str>>,
        starts: AtomicUsize,
        stops: AtomicUsize,
        closes: AtomicUsize,
        fail_start: bool,
        fail_first_stop: bool,
        start_gate: Option<(Arc<Notify>, Arc<Notify>)>,
        stop_gate: Option<(Arc<Notify>, Arc<Notify>)>,
    }

    impl Probe {
        fn new(manager: &Arc<Mutex<ChannelManager>>) -> Arc<Self> {
            Arc::new(Self {
                manager: Arc::downgrade(manager),
                calls: Mutex::new(Vec::new()),
                starts: AtomicUsize::new(0),
                stops: AtomicUsize::new(0),
                closes: AtomicUsize::new(0),
                fail_start: false,
                fail_first_stop: false,
                start_gate: None,
                stop_gate: None,
            })
        }
        async fn callback(&self, operation: &'static str) -> Result<()> {
            if let Some(manager) = self.manager.upgrade() {
                let guard = manager.lock().await;
                assert!(!guard.channel_ids().is_empty());
                for id in guard.channel_ids() {
                    let plugin = guard
                        .get(id)
                        .ok_or_else(|| channel_error("missing registered plugin".into()))?;
                    assert!(!plugin.label().is_empty());
                    assert_eq!(plugin.capabilities().chat_types, &[ChatType::Direct]);
                }
            }
            self.calls.lock().await.push(operation);
            Ok(())
        }
    }

    struct ProbePlugin {
        id: &'static str,
        probe: Arc<Probe>,
    }
    #[async_trait]
    impl ChannelPlugin for ProbePlugin {
        fn id(&self) -> &str {
            self.id
        }
        fn capabilities(&self) -> &ChannelCapabilities {
            &CAPS
        }
        async fn start(&mut self, _handler: Arc<dyn MessageHandler>) -> Result<()> {
            self.probe.starts.fetch_add(1, Ordering::AcqRel);
            self.probe.callback("start").await?;
            if let Some((entered, release)) = &self.probe.start_gate {
                entered.notify_one();
                release.notified().await;
            }
            if self.probe.fail_start {
                Err(channel_error("probe start failure".into()))
            } else {
                Ok(())
            }
        }
        async fn stop(&mut self) -> Result<()> {
            let attempt = self.probe.stops.fetch_add(1, Ordering::AcqRel);
            self.probe.callback("stop").await?;
            if let Some((entered, release)) = &self.probe.stop_gate {
                entered.notify_one();
                release.notified().await;
            }
            if self.probe.fail_first_stop && attempt == 0 {
                Err(channel_error("probe stop failure".into()))
            } else {
                Ok(())
            }
        }
        async fn send(&self, _message: OutboundMessage) -> Result<()> {
            self.probe.callback("send").await
        }
        async fn health_check(&self) -> Result<()> {
            self.probe.callback("health").await
        }
    }
    struct ProbeHandler(Arc<Probe>);
    #[async_trait]
    impl MessageHandler for ProbeHandler {
        async fn handle(&self, _message: InboundMessage) -> Result<OutboundMessage> {
            Err(channel_error("unused handle".into()))
        }
        async fn reply(&self, _message: OutboundMessage) -> Result<()> {
            Ok(())
        }
        async fn close(&self) -> Result<()> {
            self.0.closes.fetch_add(1, Ordering::AcqRel);
            self.0.callback("close").await
        }
    }

    async fn registered(
        manager: &Arc<Mutex<ChannelManager>>,
        id: &'static str,
        probe: Arc<Probe>,
    ) -> Result<ChannelHandle> {
        let mut guard = manager.lock().await;
        guard.register(Box::new(ProbePlugin { id, probe }))?;
        guard
            .get_owned(id)
            .ok_or_else(|| channel_error("registration missing".into()))
    }

        assert!(manager.get_mut("retained").is_some());
        channel.stop().await?;
        assert_eq!(probe.stops.load(Ordering::Acquire), 2);
        let health = channel.health_check();
        drop(manager);
        health.await?;
        channel
            .start(Arc::new(ProbeHandler(Arc::clone(&probe))))
            .await?;
        channel.stop().await?;
        assert_eq!(probe.starts.load(Ordering::Acquire), 2);
        assert_eq!(probe.closes.load(Ordering::Acquire), 2);
        Ok(())
    }

    #[tokio::test]
    async fn concurrent_mutable_borrow_cannot_be_cleared_by_inflight_stop_and_epoch_saturates()
    -> Result<()> {
        let manager = Arc::new(Mutex::new(ChannelManager::new()));
        let entered = Arc::new(Notify::new());
        let release = Arc::new(Notify::new());
        let mut probe = Probe::new(&manager);
        Arc::get_mut(&mut probe)
            .ok_or_else(|| channel_error("probe shared".into()))?
            .stop_gate = Some((Arc::clone(&entered), Arc::clone(&release)));
        let channel = registered(&manager, "epoch", Arc::clone(&probe)).await?;
        let first = tokio::spawn(channel.stop());
        entered.notified().await;
        assert!(manager.lock().await.get_mut("epoch").is_some());
        release.notify_one();
        first
            .await
            .map_err(|error| channel_error(error.to_string()))??;
        let second = tokio::spawn(channel.stop());
        entered.notified().await;
        release.notify_one();
        second
            .await
            .map_err(|error| channel_error(error.to_string()))??;
        assert_eq!(probe.stops.load(Ordering::Acquire), 2);
        channel
            .entry
            .mutation_epoch
            .store(u64::MAX, Ordering::Release);
        assert!(manager.lock().await.get_mut("epoch").is_some());
        assert_eq!(
            channel.entry.mutation_epoch.load(Ordering::Acquire),
            u64::MAX
        );
        for _ in 0..2 {
            let task = tokio::spawn(channel.stop());
            entered.notified().await;
            release.notify_one();
            task.await
                .map_err(|error| channel_error(error.to_string()))??;
        }
        assert_eq!(probe.stops.load(Ordering::Acquire), 4);
        Ok(())
    }

    #[tokio::test]
    async fn registration_metadata_is_fixed_while_async_calls_reach_original_plugin() -> Result<()>
    {
        struct DynamicDeclaration {
            started: bool,
        }
        #[async_trait]
        impl ChannelPlugin for DynamicDeclaration {
            fn id(&self) -> &str {
                "declared"
            }
            fn label(&self) -> &str {
                if self.started {
                    "changed"
                } else {
                    "registered"
                }
            }
            fn capabilities(&self) -> &ChannelCapabilities {
                static CHANGED: ChannelCapabilities = ChannelCapabilities {
                    chat_types: &[ChatType::Group],
                    supports_media: true,
                    supports_threads: true,
                };
                if self.started { &CHANGED } else { &CAPS }
            }
            async fn start(&mut self, _handler: Arc<dyn MessageHandler>) -> Result<()> {
                self.started = true;
                Ok(())
            }
            async fn stop(&mut self) -> Result<()> {
                Ok(())
            }
            async fn send(&self, _message: OutboundMessage) -> Result<()> {
                if self.started {
                    Ok(())
                } else {
                    Err(channel_error("original instance not started".into()))
                }
            }
        }
        let outside = Arc::new(Mutex::new(ChannelManager::new()));
        let probe = Probe::new(&outside);
        drop(outside);
        let mut manager = ChannelManager::new();
        manager.register(Box::new(DynamicDeclaration { started: false }))?;
        let channel = manager
            .get_owned("declared")
            .ok_or_else(|| channel_error("missing declaration".into()))?;
        let borrowed = manager
            .get_mut("declared")
            .ok_or_else(|| channel_error("missing mutable declaration".into()))?;
        borrowed.start(Arc::new(ProbeHandler(probe))).await?;
        assert_eq!(borrowed.label(), "registered");
        assert_eq!(borrowed.capabilities().chat_types, &[ChatType::Direct]);
        assert!(!borrowed.capabilities().supports_media);
        assert_eq!(channel.label(), "registered");
        channel
            .send(OutboundMessage::new(
                "declared",
                "chat",
                ChatType::Direct,
                "original",
            ))
            .await?;
        channel.stop().await?;
        Ok(())
    }
